🌱 1. Conferindo o sistema completo

Nosso GlowUP Plant utilizará três dispositivos conectados ao ESP32-C3: sensor de umidade, display OLED e buzzer ativo.

Componente Conexão Função
Sensor — AOUT GPIO1 Medição analógica
Sensor — VCC/GND 3V3/GND Alimentação
OLED — SDA GPIO8 Dados I²C
OLED — SCL/SCK GPIO9 Clock I²C
OLED — VCC/GND 3V3/GND Alimentação
Buzzer ativo — sinal GPIO4 Alerta sonoro
Buzzer ativo — GND GND Referência elétrica

Procedimento de montagem

  1. Desconecte o cabo USB-C.
  2. Confira as conexões do sensor.
  3. Confira SDA e SCL do OLED.
  4. Confira a polaridade do buzzer.
  5. Peça à monitora para verificar o circuito.
  6. Conecte o cabo USB-C após a conferência.
⚠️ Cuidados importantes

Os GPIOs do ESP32-C3 operam com níveis lógicos de 3,3 V.

O acionamento direto pelo GPIO4 somente é adequado se o buzzer ativo for compatível com 3,3 V e consumir corrente dentro do limite seguro do pino. Caso contrário, utilize um circuito acionador apropriado, conferido pela equipe do laboratório.

Os GPIOs 8 e 9 também participam das condições de inicialização do ESP32-C3. Confirme que o circuito inicia corretamente com o OLED conectado.

// GlowUP Plant — Etapa 05
// Experimento 2: sensor e OLED integrados.

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

const int PINO_SENSOR = 1;
const int PINO_SDA = 8;
const int PINO_SCL = 9;

// Referencias dos nossos testes.
// Substitua se o sensor for diferente.
int valorSeco = 3800;
int valorUmido = 1326;

Adafruit_SSD1306 display(128, 64, &Wire, -1);

void setup() {

  Serial.begin(115200);

  // Configura o conversor analogico.
  analogReadResolution(12);

  // Inicializa a comunicacao I2C.
  Wire.begin(PINO_SDA, PINO_SCL);

  // Inicializa o display.
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("OLED nao encontrado!");
    while (true) {
      delay(1000);
    }
  }

  display.setTextColor(SSD1306_WHITE);
}

void loop() {

  // Le o valor do sensor.
  int leitura = analogRead(PINO_SENSOR);

  // Verifica a calibracao.
  if (valorSeco == valorUmido) {
    Serial.println("Erro de calibracao!");
    delay(1000);
    return;
  }

  // Calcula a porcentagem relativa.
  int umidade = map(
    leitura, valorSeco, valorUmido, 0, 100
  );

  // Mantem o valor entre 0 e 100%.
  umidade = constrain(umidade, 0, 100);

  // Limpa a imagem anterior.
  display.clearDisplay();

  // Titulo superior.
  display.setTextSize(1);
  display.setCursor(28, 2);
  display.println("GlowUP Plant");

  // Identificacao da leitura.
  display.setCursor(19, 20);
  display.println("Umidade do solo");

  // Porcentagem em destaque.
  display.setTextSize(2);
  display.setCursor(40, 32);
  display.print(umidade);
  display.print("%");

  // Mostra a condicao do solo.
  display.setTextSize(1);
  display.setCursor(16, 55);

  if (umidade < 30) {
    display.println("Solo seco!");
  }
  else if (umidade < 70) {
    display.println("Solo adequado");
  }
  else {
    display.setCursor(2, 55);
    display.println("Solo muito umido");
  }

  // Atualiza a tela.
  display.display();

  // Tambem envia a leitura ao computador.
  Serial.print("Umidade: ");
  Serial.print(umidade);
  Serial.println("%");

  delay(1000);
}

🌱 3. Criando um alerta sonoro

Quando a planta precisar de água, o GlowUP Plant poderá emitir um aviso sonoro.

O buzzer deverá produzir dois bipes curtos quando a umidade estiver abaixo de 30%.

Funcionamento desejado

Umidade Buzzer
Abaixo de 30% Dois bipes curtos
30% ou mais Permanece desligado 
// GlowUP Plant — Experimento 3
// Objetivo: produzir um alerta quando
// a umidade do solo estiver baixa.

const int PINO_SENSOR = 1;
const int PINO_BUZZER = 4;

// Valores de calibracao.
int valorSeco = 3800;
int valorUmido = 1326;

// Limite didatico para solo seco.
const int LIMITE_SECO = 30;

// Funcao que produz dois bipes curtos.
void emitirAlerta() {

  // Repete duas vezes.
  for (int i = 0; i < 2; i++) {

    // Liga o buzzer.
    digitalWrite(PINO_BUZZER, HIGH);

    // Mantem ligado por 150 ms.
    delay(150);

    // Desliga o buzzer.
    digitalWrite(PINO_BUZZER, LOW);

    // Pausa entre os bipes.
    delay(200);
  }
}

void setup() {

  Serial.begin(115200);
  analogReadResolution(12);

  // Configura o buzzer como saida.
  pinMode(PINO_BUZZER, OUTPUT);

  // Inicia desligado.
  digitalWrite(PINO_BUZZER, LOW);
}

void loop() {

  int leitura = analogRead(PINO_SENSOR);

  if (valorSeco == valorUmido) {
    Serial.println("Erro de calibracao!");
    delay(1000);
    return;
  }

  int umidade = map(
    leitura, valorSeco, valorUmido, 0, 100
  );

  umidade = constrain(umidade, 0, 100);

  Serial.print("Umidade: ");
  Serial.print(umidade);
  Serial.println("%");

  // Verifica se o solo esta seco.
  if (umidade < LIMITE_SECO) {

    Serial.println("Preciso de agua!");

    // Executa o alerta sonoro.
    emitirAlerta();
  }
  else {

    Serial.println("Umidade adequada.");
  }

  // Aguarda cinco segundos.
  delay(5000);
}
🔊 Atenção ao som!

O programa repetirá o alerta enquanto a umidade continuar abaixo do limite.

Para evitar alarmes repetitivos no projeto final, criaremos um intervalo maior entre os avisos.

Utilize apenas um buzzer ativo compatível com o circuito de acionamento.

🌱 5. Construindo nosso GlowUP Plant!

Chegamos ao programa principal! Vamos reunir os recursos desenvolvidos anteriormente.

O GlowUP Plant apresentará:

  • Porcentagem de umidade.
  • Expressão facial correspondente.
  • Alerta sonoro para solo seco.
  • Intervalo mínimo entre alertas.
💡 Uma melhoria importante

O buzzer não deve tocar continuamente enquanto o solo permanecer seco.

Utilizaremos a função millis() para controlar o tempo entre alertas.

Neste exemplo, o sistema repetirá o aviso, no máximo, uma vez a cada 30 segundos.

Programa principal

// ==========================================
// GlowUP Plant — Etapa 05
// Experimento 5: sistema integrado
// Versao didatica 1.0
// ==========================================

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// Pinos utilizados.
const int PINO_SENSOR = 1;
const int PINO_BUZZER = 4;
const int PINO_SDA = 8;
const int PINO_SCL = 9;

// Referencias de calibracao.
int valorSeco = 3800;
int valorUmido = 1326;

// Limites didaticos de umidade.
const int LIMITE_SECO = 30;
const int LIMITE_UMIDO = 70;

// Intervalo minimo entre alertas.
// 30000 milissegundos = 30 segundos.
const unsigned long INTERVALO_ALERTA = 30000;

// Guarda o instante do ultimo alerta.
unsigned long ultimoAlerta = 0;

// Indica se ja houve um alerta
// durante a condicao atual de solo seco.
bool alertaEnviado = false;

// Objeto do display.
Adafruit_SSD1306 display(128, 64, &Wire, -1);


// ==========================================
// FUNCOES DE DESENHO
// ==========================================

void desenharRosto() {
  display.drawCircle(64, 40, 20, SSD1306_WHITE);
  display.fillCircle(56, 34, 2, SSD1306_WHITE);
  display.fillCircle(72, 34, 2, SSD1306_WHITE);
}

void rostoFeliz() {
  desenharRosto();
  display.drawLine(53,45,59,50,SSD1306_WHITE);
  display.drawLine(59,50,69,50,SSD1306_WHITE);
  display.drawLine(69,50,75,45,SSD1306_WHITE);
}

void rostoTriste() {
  desenharRosto();
  display.drawLine(53,51,59,46,SSD1306_WHITE);
  display.drawLine(59,46,69,46,SSD1306_WHITE);
  display.drawLine(69,46,75,51,SSD1306_WHITE);
}

void rostoSurpreso() {
  desenharRosto();
  display.drawCircle(64,49,5,SSD1306_WHITE);
}


// ==========================================
// FUNCAO DE ALERTA SONORO
// ==========================================

void emitirAlerta() {

  // Produz dois bipes curtos.
  for (int i = 0; i < 2; i++) {

    digitalWrite(PINO_BUZZER, HIGH);
    delay(150);

    digitalWrite(PINO_BUZZER, LOW);
    delay(200);
  }
}


// ==========================================
// MOSTRAR UMIDADE NO OLED
// ==========================================

void mostrarUmidade(int umidade) {

  display.clearDisplay();

  // Titulo na faixa superior.
  display.setTextSize(1);
  display.setCursor(28, 2);
  display.println("GlowUP Plant");

  // Identificacao.
  display.setCursor(19, 20);
  display.println("Umidade do solo");

  // Porcentagem em destaque.
  display.setTextSize(2);
  display.setCursor(40, 32);
  display.print(umidade);
  display.print("%");

  // Estado do solo.
  display.setTextSize(1);
  display.setCursor(2, 55);

  if (umidade < LIMITE_SECO) {
    display.println("Solo seco!");
  }
  else if (umidade < LIMITE_UMIDO) {
    display.println("Solo adequado");
  }
  else {
    display.println("Solo muito umido");
  }

  display.display();
}


// ==========================================
// MOSTRAR EXPRESSAO FACIAL
// ==========================================

void mostrarExpressao(int umidade) {

  display.clearDisplay();

  // Titulo.
  display.setTextSize(1);
  display.setCursor(28, 2);
  display.println("GlowUP Plant");

  // Seleciona o rosto.
  if (umidade < LIMITE_SECO) {
    rostoTriste();
  }
  else if (umidade < LIMITE_UMIDO) {
    rostoFeliz();
  }
  else {
    rostoSurpreso();
  }

  display.display();
}


// ==========================================
// CONFIGURACAO INICIAL
// ==========================================

void setup() {

  Serial.begin(115200);
  delay(1500);

  // ADC de 12 bits.
  analogReadResolution(12);

  // Configura o buzzer.
  pinMode(PINO_BUZZER, OUTPUT);
  digitalWrite(PINO_BUZZER, LOW);

  // Inicializa o I2C.
  Wire.begin(PINO_SDA, PINO_SCL);

  // Inicializa o display.
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {

    Serial.println("OLED nao encontrado!");

    while (true) {
      delay(1000);
    }
  }

  display.setTextColor(SSD1306_WHITE);

  Serial.println("GlowUP Plant iniciado!");
}


// ==========================================
// PROGRAMA PRINCIPAL
// ==========================================

void loop() {

  // 1. Leitura do sensor.
  int leitura = analogRead(PINO_SENSOR);

  // Verifica as referencias.
  if (valorSeco == valorUmido) {

    Serial.println("Erro de calibracao!");

    digitalWrite(PINO_BUZZER, LOW);

    delay(1000);
    return;
  }

  // 2. Calculo da umidade percentual.
  int umidade = map(
    leitura,
    valorSeco,
    valorUmido,
    0,
    100
  );

  umidade = constrain(umidade, 0, 100);

  // 3. Mostra os valores no Monitor Serial.
  Serial.print("Leitura ADC: ");
  Serial.println(leitura);

  Serial.print("Umidade: ");
  Serial.print(umidade);
  Serial.println("%");

  // 4. Mostra a porcentagem no OLED.
  mostrarUmidade(umidade);

  // Mantem a tela por dois segundos.
  delay(2000);

  // 5. Mostra a expressao da planta.
  mostrarExpressao(umidade);

  // 6. Verifica a necessidade de alerta.
  if (umidade < LIMITE_SECO) {

    // Consulta o tempo atual.
    unsigned long agora = millis();

    // Emite alerta imediatamente na
    // primeira deteccao de solo seco.
    // Depois respeita o intervalo definido.
    if (
      !alertaEnviado ||
      agora - ultimoAlerta >= INTERVALO_ALERTA
    ) {

      emitirAlerta();

      // Registra o instante do alerta.
      ultimoAlerta = millis();

      // Marca que o alerta ja ocorreu.
      alertaEnviado = true;
    }
  }
  else {

    // Se o solo deixou de estar seco,
    // libera um novo alerta imediato
    // em uma futura condicao de solo seco.
    alertaEnviado = false;

    // Garante buzzer desligado.
    digitalWrite(PINO_BUZZER, LOW);
  }

  // Mantem o rosto visivel por dois segundos.
  delay(2000);
}
