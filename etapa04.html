🎨 Como é o OLED da oficina?

Utilizaremos o display OLED SSD1306 de 128 × 64 pixels.

Algumas placas possuem pixels brancos. Outras possuem uma faixa superior amarela e uma região inferior azul.

O programa é o mesmo nos dois modelos. A diferença está na cor física dos pixels.

💡 Nossa regra de organização

Reservaremos as linhas de 0 a 15 para o título e as linhas de 16 a 63 para números e desenhos.

Dessa maneira, o mesmo programa terá uma apresentação organizada tanto no OLED branco quanto no bicolor.

🌱 1. Conhecendo e conectando o OLED

O OLED é uma pequena tela que permite mostrar textos, números e figuras.

Seu controlador é o SSD1306 e a comunicação será realizada pelo barramento I²C.

Terminal OLED ESP32-C3 Função
VCC 3V3 Alimentação
GND GND Referência elétrica
SDA GPIO8 Dados I²C
SCL / SCK GPIO9 Clock I²C
🧰 Bibliotecas necessárias

No Gerenciador de Bibliotecas do Arduino IDE, localize:

  • Adafruit SSD1306
  • Adafruit GFX Library

A biblioteca Wire é utilizada para comunicação I²C.

A equipe do laboratório deve utilizar versões das bibliotecas compatíveis com o Arduino IDE instalado.

⚠️ Confira a montagem

Desconecte o USB antes de modificar qualquer conexão.

Os GPIOs 8 e 9 participam também das condições de inicialização do ESP32-C3. A ligação I²C escolhida deve ser validada na placa real.

Vamos escrever nossa primeira mensagem diretamente na tela!

// GlowUP Plant — Experimento 2
// Objetivo: apresentar nossa primeira mensagem.

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// Dimensoes do display.
#define LARGURA 128
#define ALTURA 64

// Conexoes I2C.
const int PINO_SDA = 8;
const int PINO_SCL = 9;

// Cria o objeto display.
// -1 indica ausencia de reset dedicado.
Adafruit_SSD1306 display(128, 64, &Wire, -1);

void setup() {

  // Inicia o barramento I2C.
  Wire.begin(PINO_SDA, PINO_SCL);

  // Inicializa o OLED no endereco 0x3C.
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    while (true) {
      delay(1000);
    }
  }

  // Limpa a memoria grafica.
  display.clearDisplay();

  // Define a cor dos pixels acesos.
  display.setTextColor(SSD1306_WHITE);

  // Titulo na faixa superior.
  display.setTextSize(1);
  display.setCursor(28, 2);
  display.println("GlowUP Plant");

  // Mensagem principal, com letras maiores.
  display.setTextSize(2);
  display.setCursor(40, 27);
  display.println("OLA!");

  // Mensagem inferior.
  display.setTextSize(1);
  display.setCursor(2, 55);
  display.println("Minha primeira tela!");

  // Atualiza os pixels da tela.
  display.display();
}

void loop() {
  // A mensagem permanece exibida.
}

// GlowUP Plant — Experimento 3
// Objetivo: mostrar a leitura atual no OLED.

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

const int PINO_SDA = 8;
const int PINO_SCL = 9;
const int PINO_SENSOR = 1;

Adafruit_SSD1306 display(128, 64, &Wire, -1);

void setup() {

  Serial.begin(115200);

  // Configura o ADC para 12 bits.
  analogReadResolution(12);

  // Inicia a comunicacao I2C.
  Wire.begin(PINO_SDA, PINO_SCL);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("OLED nao encontrado!");
    while (true) {
      delay(1000);
    }
  }

  display.setTextColor(SSD1306_WHITE);
}

void loop() {

  // Obtem a leitura do sensor.
  int leitura = analogRead(PINO_SENSOR);

  // Limpa a imagem anterior.
  display.clearDisplay();

  // Titulo.
  display.setTextSize(1);
  display.setCursor(28, 2);
  display.println("GlowUP Plant");

  // Identificacao da leitura.
  display.setCursor(10, 20);
  display.println("Leitura atual:");

  // Numero em tamanho maior.
  display.setTextSize(2);
  display.setCursor(40, 34);
  display.println(leitura);

  // Atualiza a tela.
  display.display();

  // Apresenta tambem no Monitor Serial.
  Serial.println(leitura);

  delay(1000);
}

🌱 4. Leitura atual e média

Vamos calcular a média de cinco leituras consecutivas e apresentar os dois valores no display.

Resultado esperado

// GlowUP Plant — Experimento 4
// Objetivo: mostrar leitura atual e media.

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

const int PINO_SDA = 8;
const int PINO_SCL = 9;
const int PINO_SENSOR = 1;

const int NUM_LEITURAS = 5;

Adafruit_SSD1306 display(128, 64, &Wire, -1);

void setup() {

  Serial.begin(115200);
  analogReadResolution(12);

  Wire.begin(PINO_SDA, PINO_SCL);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    while (true) {
      delay(1000);
    }
  }

  display.setTextColor(SSD1306_WHITE);
}

void loop() {

  // Leitura atual.
  int atual = analogRead(PINO_SENSOR);

  // Acumulador das cinco leituras.
  long soma = 0;

  // Executa cinco leituras consecutivas.
  for (int i = 0; i < NUM_LEITURAS; i++) {

    soma = soma + analogRead(PINO_SENSOR);

    delay(50);
  }

  // Calcula a media.
  float media = soma / float(NUM_LEITURAS);

  display.clearDisplay();

  display.setTextSize(1);

  display.setCursor(28, 2);
  display.println("GlowUP Plant");

  display.setCursor(2, 20);
  display.print("Atual: ");
  display.println(atual);

  display.setCursor(2, 35);
  display.print("Media: ");
  display.println(media, 0);

  display.setCursor(2, 54);
  display.println("Monitorando...");

  display.display();

  Serial.print("Atual: ");
  Serial.println(atual);

  Serial.print("Media: ");
  Serial.println(media);

  delay(1000);
}

🌱 5. Umidade percentual no OLED

Vamos transformar os valores do sensor em uma escala relativa de 0% a 100%.

Resultado esperado

Utilizando nossas medições

As medições podem ser recuperadas da Etapa 03.

Antes de gravar:

Os códigos abaixo usam inicialmente as referências 3800 e 1326. Confirme se são adequadas ao seu sensor.

Os campos acima registram os valores neste navegador, mas não modificam automaticamente o programa Arduino. 

// GlowUP Plant — Experimento 5
// Objetivo: apresentar a umidade percentual.

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

const int PINO_SDA = 8;
const int PINO_SCL = 9;
const int PINO_SENSOR = 1;

// Referencias de calibracao.
// Ajuste para o sensor utilizado.
int valorSeco = 3800;
int valorUmido = 1326;

Adafruit_SSD1306 display(128, 64, &Wire, -1);

void setup() {

  Serial.begin(115200);
  analogReadResolution(12);

  Wire.begin(PINO_SDA, PINO_SCL);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    while (true) {
      delay(1000);
    }
  }

  display.setTextColor(SSD1306_WHITE);
}

void loop() {

  // Le o sinal analogico.
  int leitura = analogRead(PINO_SENSOR);

  // Evita usar duas referencias iguais.
  if (valorSeco == valorUmido) {
    Serial.println("Erro de calibracao!");
    delay(1000);
    return;
  }

  // Converte para uma escala relativa.
  int umidade = map(
    leitura, valorSeco, valorUmido, 0, 100
  );

  // Limita o resultado de 0 a 100.
  umidade = constrain(umidade, 0, 100);

  display.clearDisplay();

  // Titulo na faixa superior.
  display.setTextSize(1);
  display.setCursor(28, 2);
  display.println("GlowUP Plant");

  // Identificacao.
  display.setCursor(19, 20);
  display.println("Umidade do solo");

  // Porcentagem em destaque.
  display.setTextSize(2);
  display.setCursor(40, 32);
  display.print(umidade);
  display.print("%");

  // Texto na parte inferior.
  display.setTextSize(1);
  display.setCursor(16, 55);

  if (umidade < 30) {
    display.println("Solo seco!");
  }
  else if (umidade < 70) {
    display.println("Solo adequado");
  }
  else {
    display.println("Solo muito umido");
  }

  display.display();

  Serial.print("Umidade: ");
  Serial.print(umidade);
  Serial.println("%");

  delay(1000);
}
💡 O que aprendemos?

A função map() transforma a leitura em uma escala relativa de porcentagem.

Já a função constrain() impede valores menores que zero ou maiores que 100.

🎨 Aprendendo a desenhar no OLED

Antes de começar os desafios avançados, vamos descobrir como utilizar as funções gráficas da biblioteca Adafruit GFX.

Pense no display como uma folha de papel formada por pequenos pontos. Cada ponto é chamado de pixel.

1. Entendendo as coordenadas

Nosso OLED possui 128 pixels na horizontal e 64 na vertical.

X: posição horizontal, da esquerda para a direita. Vai de 0 até 127.

Y: posição vertical, de cima para baixo. Vai de 0 até 63.

A origem (0,0) está no canto superior esquerdo, e não no canto inferior, como em muitos gráficos matemáticos.

2. Desenhando um retângulo

Utilizamos a função drawRect(). 

display.drawRect(
  x, y, largura, altura, SSD1306_WHITE
);

Entendendo os parâmetros

x, y: posição do canto superior esquerdo.

largura: tamanho horizontal em pixels.

altura: tamanho vertical em pixels.

Exemplo:

display.drawRect(
  10, 38, 104, 14, SSD1306_WHITE
);

Isso desenha um retângulo começando na posição (10,38), com largura de 104 pixels e altura de 14 pixels.

3. Preenchendo um retângulo

Para desenhar um retângulo preenchido, utilizamos fillRect(). 

// Contorno.
display.drawRect(
  10, 38, 104, 14, SSD1306_WHITE
);

// Preenchimento.
display.fillRect(
  12, 40, 50, 10, SSD1306_WHITE
);

Observe: o preenchimento começa dois pixels à direita e dois pixels abaixo do contorno.

Isso deixa uma pequena margem interna e evita sobrepor as bordas da barra.

O valor 50 corresponde à largura preenchida. Podemos modificá-lo para representar diferentes porcentagens.

4. Desenhando círculos

Para criar olhos, rostos e outras figuras, podemos utilizar círculos.

Para desenhar apenas o contorno: 

display.drawCircle(
  64, 40, 20, SSD1306_WHITE
);

Para desenhar um círculo preenchido: 

display.fillCircle(
  56, 34, 2, SSD1306_WHITE
);

Os dois primeiros números representam o centro do círculo. O terceiro representa o raio, em pixels.

5. Desenhando linhas

Uma linha é definida pelas coordenadas de seus dois extremos. 

display.drawLine(
  x1, y1, x2, y2, SSD1306_WHITE
);

Essa função será útil para desenhar as bocas das expressões faciais.

6. Desenhando triângulos

Um triângulo é definido por três pontos. Para preenchê-lo, utilizamos fillTriangle(). 

display.fillTriangle(
  45, 35,
  83, 35,
  64, 58,
  SSD1306_WHITE
);

Utilizaremos dois círculos e um triângulo para formar o coração do próximo desafio.

⚠️ Uma instrução indispensável

Depois de desenhar na memória do display, utilize: 

display.display();

Sem essa instrução, os novos desenhos poderão não aparecer na tela.

Ao iniciar um novo desenho, normalmente utilizamos também: 

display.clearDisplay();

💡 Como calcular larguraBarra?

O espaço útil para o preenchimento possui 100 pixels.

Portanto: 0% → 0 pixels, 50% → 50 pixels, 100% → 100 pixels.

// GlowUP Plant — Desafio C
// Objetivo: mostrar as emocoes da planta.

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

const int PINO_SDA = 8;
const int PINO_SCL = 9;
const int PINO_SENSOR = 1;

// Valores de calibracao.
int valorSeco = 3800;
int valorUmido = 1326;

Adafruit_SSD1306 display(128, 64, &Wire, -1);

// Desenha o contorno e os olhos.
void desenharRosto() {

  display.drawCircle(64, 40, 20, SSD1306_WHITE);

  display.fillCircle(56, 34, 2, SSD1306_WHITE);
  display.fillCircle(72, 34, 2, SSD1306_WHITE);
}

// Expressao feliz.
void rostoFeliz() {

  desenharRosto();

  display.drawLine(53,45,59,50,SSD1306_WHITE);
  display.drawLine(59,50,69,50,SSD1306_WHITE);
  display.drawLine(69,50,75,45,SSD1306_WHITE);
}

// Expressao triste.
void rostoTriste() {

  desenharRosto();

  display.drawLine(53,51,59,46,SSD1306_WHITE);
  display.drawLine(59,46,69,46,SSD1306_WHITE);
  display.drawLine(69,46,75,51,SSD1306_WHITE);
}

// Expressao surpresa.
void rostoSurpreso() {

  desenharRosto();

  display.drawCircle(64,49,5,SSD1306_WHITE);
}

void setup() {

  Serial.begin(115200);
  analogReadResolution(12);

  Wire.begin(PINO_SDA, PINO_SCL);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    while (true) {
      delay(1000);
    }
  }

  display.setTextColor(SSD1306_WHITE);
}

void loop() {

  int leitura = analogRead(PINO_SENSOR);

  if (valorSeco == valorUmido) {
    Serial.println("Erro de calibracao!");
    delay(1000);
    return;
  }

  int umidade = map(
    leitura, valorSeco, valorUmido, 0, 100
  );

  umidade = constrain(umidade, 0, 100);

  display.clearDisplay();

  // Titulo superior.
  display.setTextSize(1);
  display.setCursor(28, 2);
  display.println("GlowUP Plant");

  // TODO 1:
  // Se a umidade for menor que 30,
  // execute rostoTriste().


  // TODO 2:
  // Se a umidade for menor que 70,
  // execute rostoFeliz().


  // TODO 3:
  // Nos demais casos,
  // execute rostoSurpreso().


  // Atualiza a imagem.
  display.display();

  delay(1000);
}
// GlowUP Plant — Solucao do Desafio C

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

const int PINO_SDA = 8;
const int PINO_SCL = 9;
const int PINO_SENSOR = 1;

int valorSeco = 3800;
int valorUmido = 1326;

Adafruit_SSD1306 display(128, 64, &Wire, -1);

// Contorno e olhos.
void desenharRosto() {

  display.drawCircle(64, 40, 20, SSD1306_WHITE);

  display.fillCircle(56, 34, 2, SSD1306_WHITE);
  display.fillCircle(72, 34, 2, SSD1306_WHITE);
}

// Boca sorrindo.
void rostoFeliz() {

  desenharRosto();

  display.drawLine(53,45,59,50,SSD1306_WHITE);
  display.drawLine(59,50,69,50,SSD1306_WHITE);
  display.drawLine(69,50,75,45,SSD1306_WHITE);
}

// Boca triste.
void rostoTriste() {

  desenharRosto();

  display.drawLine(53,51,59,46,SSD1306_WHITE);
  display.drawLine(59,46,69,46,SSD1306_WHITE);
  display.drawLine(69,46,75,51,SSD1306_WHITE);
}

// Boca de surpresa.
void rostoSurpreso() {

  desenharRosto();

  display.drawCircle(64,49,5,SSD1306_WHITE);
}

void setup() {

  Serial.begin(115200);
  analogReadResolution(12);

  Wire.begin(PINO_SDA, PINO_SCL);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    while (true) {
      delay(1000);
    }
  }

  display.setTextColor(SSD1306_WHITE);
}

void loop() {

  int leitura = analogRead(PINO_SENSOR);

  if (valorSeco == valorUmido) {
    Serial.println("Erro de calibracao!");
    delay(1000);
    return;
  }

  int umidade = map(
    leitura, valorSeco, valorUmido, 0, 100
  );

  umidade = constrain(umidade, 0, 100);

  display.clearDisplay();

  // Titulo superior.
  display.setTextSize(1);
  display.setCursor(28, 2);
  display.println("GlowUP Plant");

  // Escolhe a expressao conforme a umidade.
  if (umidade < 30) {

    rostoTriste();

  }
  else if (umidade < 70) {

    rostoFeliz();

  }
  else {

    rostoSurpreso();

  }

  display.display();

  Serial.print("Umidade: ");
  Serial.print(umidade);
  Serial.println("%");

  delay(1000);
}
